cinema/catalogo_filmes.c 
#include <stdio.h>

// Um filme do catálogo
typedef struct {
  char titulo[40];
  int ano;
  float nota;
} Filme;

int main(void) {
  Filme lista[3] = {
    {"Blade Runner 2049", 2017, 9.0},
    {"2001: Uma Odisseia", 1968, 9.5},
    {"Oppenheimer", 2023, 8.8}
  };
  float soma = 0;

  for (int i = 0; i < 3; i++) {
    printf("%s (%d) - nota %.1f\n", lista[i].titulo, lista[i].ano, lista[i].nota);
    soma += lista[i].nota;
  }

  printf("Media: %.1f\n", soma / 3);
  return 0;
}
